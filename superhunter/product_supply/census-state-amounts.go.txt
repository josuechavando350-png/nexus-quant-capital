package main

import (
	"encoding/json"
	"fmt"
	"os"
	"strconv"

	sdkmath "cosmossdk.io/math"
)

type row struct {
	Text       string `json:"text"`
	Uint       bool   `json:"uint"`
	Canonical  string `json:"canonical"`
	Positive   bool   `json:"positive"`
	Bits       int    `json:"bits"`
}

func main() {
	var values []string
	if err := json.NewDecoder(os.Stdin).Decode(&values); err != nil { panic(err) }
	rows := make([]row, 0, len(values))
	for _, text := range values {
		result := row{Text: text}
		if signed, ok := sdkmath.NewIntFromString(text); ok { result.Positive = signed.IsPositive() }
		value, err := sdkmath.ParseUint(text)
		if err == nil {
			raw, err := value.MarshalJSON()
			if err != nil { panic(err) }
			var decoded sdkmath.Uint
			if err := decoded.UnmarshalJSON(raw); err != nil || !decoded.Equal(value) {
				panic("SDK quantity did not round-trip")
			}
			normalized, err := strconv.Unquote(string(raw))
			if err != nil { panic(err) }
			result.Uint, result.Canonical, result.Bits = true, normalized, decoded.BigInt().BitLen()
		}
		rows = append(rows, result)
	}
	if err := json.NewEncoder(os.Stdout).Encode(map[string]any{"rows": rows}); err != nil {
		panic(fmt.Sprint(err))
	}
}
